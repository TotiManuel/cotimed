export const crearUsuario = async (data) => {

    const existente = await prisma.usuario.findUnique({
        where: {
            email: data.email
        }
    });

    if (existente) {
        throw new Error("Email ya registrado");
    }

    const usuario = await prisma.usuario.create({
        data: {
            nombre: data.nombre,
            email: data.email,
            password: data.password,
            rol: data.rol
        }
    });

    return usuario;
};
